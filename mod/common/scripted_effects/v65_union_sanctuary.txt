			}

			set_variable = {

			}

				v65_real_victories = PKS.v65_real_victories

v65_open_sanctuary = {
	if = {
		limit = {
			tag = KHM
			NOT = {
				has_war_with = SVN
			}
			NOT = {
				has_war_with = NVN
			}
			NOT = {
				has_war_with = FNL
			}
			controls_state = 35
			controls_state = 42
		}
		set_country_flag = v65_sanctuary_open
		add_ideas = v65_neutral_sanctuary
	}
}
v65_close_sanctuary = {
	clr_country_flag = v65_sanctuary_open
	remove_ideas = v65_neutral_sanctuary
}
v65_sanctuary_pulse = {
	if = {
		limit = {
			v65_sanctuary_operational = yes
		}
		NVN = {
			add_to_variable = {
				v65_trail_level = 1
			}
			clamp_variable = {
				var = v65_trail_level
				min = 0
				max = 5
			}
		}
		FNL = {
			add_to_variable = {
				v65_infiltration = 1
			}
			clamp_variable = {
				var = v65_infiltration
				min = 0
				max = 3
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment_2
				amount = 300
				producer = NVN
			}
		}
		add_to_variable = {
			v65_border_tension = 1
		}
		clamp_variable = {
			var = v65_border_tension
			min = 0
			max = 3
		}
		if = {
			limit = {
				NOT = {
					has_country_flag = v65_sanctuary_discovered
				}
			}
			set_country_flag = v65_sanctuary_discovered
			SVN = {
				country_event = {
					id = v65routes.5
					days = 1
				}
			}
		}
	}
}
