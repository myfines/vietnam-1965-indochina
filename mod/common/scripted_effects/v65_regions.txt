v65_devolve_regions = {
	if = {
		limit = {
			tag = SVN
			NOT = {
				has_country_flag = v65_regionalized
			}
		}
		set_country_flag = v65_regionalized
		add_ideas = v65_decentralized_command
		country_event = {
			id = v65regions.11
			days = 1
		}
		country_event = {
			id = v65regions.12
			days = 2
		}
		country_event = {
			id = v65regions.13
			days = 3
		}
		country_event = {
			id = v65regions.14
			days = 4
		}
	}
}
v65_regional_reintegration_HLC = {
	if = {
		limit = {
			tag = SVN
			HLC = {
				exists = yes
				is_ai = yes
				check_variable = {
					v65_regional_loyalty > 74
				}
			}
		}
		if = {
			limit = {
				has_country_leader = {
					character = v65_duong_van_minh
				}
			}
			set_country_flag = v65_preserve_minh
		}
		HLC = {
			v65_nguyen_khanh = {
				set_nationality = SVN
			}
		}
		annex_country = {
			target = HLC
			transfer_troops = yes
		}
		add_ideas = v65_reintegrated_command
		if = {
			limit = {
				has_country_flag = v65_preserve_minh
			}
			promote_character = {
				character = v65_duong_van_minh
				ideology = despotism
			}
			clr_country_flag = v65_preserve_minh
		}
		else = {
			promote_character = {
				character = v65_nguyen_van_thieu
				ideology = despotism
			}
		}
	}
}
